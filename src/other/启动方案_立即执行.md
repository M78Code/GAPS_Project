```python
import numpy as np
import struct


class DATFileReader:
   """
   读取闪烁体.dat文件的工具
   
   文件结构：
   - 每个事件包含：Header + 4通道波形数据
   - 每通道：1024个样本（通常是float32）
   """

   def __init__(self, filepath):
      self.filepath = filepath
      self.extract_label()

   def extract_label(self):
      """从文件名提取位置标签"""
      import os
      basename = os.path.basename(self.filepath)
      # run7_65.dat -> 65
      self.label = float(basename.split('_')[1].split('.')[0])

   def read_file(self):
      """
      读取整个.dat文件
      
      返回：
          events: list of dict
              - 'CH0': [1024] 波形
              - 'CH1': [1024] 波形
              - 'CH2': [1024] 波形（触发）
              - 'CH3': [1024] 波形（触发）
              - 'label': float（位置标签）
      """
      events = []

      try:
         with open(self.filepath, 'rb') as f:
            # 尝试读取（需要根据实际.dat格式调整）
            # 这里是通用的float32格式

            while True:
               # 读取4通道数据（每通道1024个float32）
               data = f.read(4 * 1024 * 4)  # 4通道 × 1024样本 × 4字节

               if not data:
                  break

               # 解析为4通道
               samples = struct.unpack(f'<{4 * 1024}f', data)

               ch0 = np.array(samples[0:1024])
               ch1 = np.array(samples[1024:2048])
               ch2 = np.array(samples[2048:3072])
               ch3 = np.array(samples[3072:4096])

               event = {
                  'CH0': ch0,
                  'CH1': ch1,
                  'CH2': ch2,
                  'CH3': ch3,
                  'label': self.label
               }
               events.append(event)

      except Exception as e:
         print(f"⚠️ 读取失败: {e}")
         print("可能需要调整文件格式解析")

      return events
```

### 5️⃣ 验证所有数据文件

**脚本**：`/Users/lind/Desktop/ppt/GAPS_Project/dataset/verify_data.py`

```python
import os
import numpy as np
from src.dat_reader import DATFileReader

raw_dir = '/Users/lind/Desktop/ppt/GAPS_Project/dataset/raw'

# 列出所有.dat文件
dat_files = sorted([f for f in os.listdir(raw_dir) if f.endswith('.dat')])

print(f"找到 {len(dat_files)} 个数据文件\n")
print("文件 | 位置标签(cm) | 事件数")
print("-" * 40)

all_labels = []
all_counts = []

for filename in dat_files:
    filepath = os.path.join(raw_dir, filename)
    
    try:
        reader = DATFileReader(filepath)
        events = reader.read_file()
        label = reader.label
        count = len(events)
        
        all_labels.append(label)
        all_counts.append(count)
        
        print(f"{filename:15} | {label:6.0f} | {count:5}")
    
    except Exception as e:
        print(f"{filename:15} | 读取失败: {e}")

print("\n" + "="*40)
print(f"位置覆盖范围: {min(all_labels):.0f} - {max(all_labels):.0f} cm")
print(f"总事件数: {sum(all_counts)}")
print(f"平均每个位置: {sum(all_counts)/len(all_counts):.0f} 个事件")
```
